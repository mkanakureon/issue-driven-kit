$upTo = 'S6b'
# （作る物は変数で指す）
$scripts = 'Scripts'; $builds = 'Builds'
# M1-1〜M1-5 と S6b: 全部のテスト
pwsh -NoProfile -File Scripts/unity-test.ps1 -Platform All -Filter 'KaedeScapeBuii.Tests'

$doc = Get-Content Docs/Guides/Working/scape-m1-detailed-design.md -Raw -Encoding utf8

$rows = @([regex]::Matches($doc, '(?m)^\| `KaedeScapeBuii\.Tests\.(EditMode|PlayMode)\.(\w+)\.(\w+)` \| (S1|S2|S4|S6a|S6b) \|'))

foreach ($m in $need) {

# M1-7: EXE を -scape-autoshot で起動し、既定の保存先 Builds/<ID>/Shots/ の 3 枚の大きさを PNG の頭で見る
function Get-PngSize([string]$p) {
    $b = [System.IO.File]::ReadAllBytes($p)
    if ($b.Length -lt 24 -or [System.Text.Encoding]::ASCII.GetString($b, 1, 3) -ne 'PNG') { return $null }
    # [byte] のまま -shl すると Byte に切り捨てられる（1920 が 128 になる）。[int] にしてからずらす
    $w = ([int]$b[16] -shl 24) -bor ([int]$b[17] -shl 16) -bor ([int]$b[18] -shl 8) -bor [int]$b[19]
    $h = ([int]$b[20] -shl 24) -bor ([int]$b[21] -shl 16) -bor ([int]$b[22] -shl 8) -bor [int]$b[23]
    return "${w}x${h}"
}

    $result = Join-Path $shots 'autoshot-result.json'
    if (-not (Test-Path $result) -or (Get-Item $result).LastWriteTime -lt $start) { Write-Host "${id}: 今回の autoshot-result.json が $shots に無い（既定の保存先: 7.2）"; exit 1 }
    if ((Get-Content $result -Raw -Encoding utf8 | ConvertFrom-Json).result -ne 'PASS') { Write-Host "${id}: autoshot-result.json の result が PASS でない"; exit 1 }
    foreach ($res in @(@('FullHD', '1920x1080'), @('2K', '2560x1440'), @('4K', '3840x2160'))) {
        $png = @(Get-ChildItem $shots -Filter "${id}_*_35mm_$($res[0]).png" | Where-Object { $_.LastWriteTime -ge $start })
        if ($png.Count -ne 1) { Write-Host "${id}: 今回撮った $($res[0]) の PNG が $($png.Count) 枚"; exit 1 }
        $size = Get-PngSize $png[0].FullName
        if ($size -ne $res[1]) { Write-Host "${id}: $($res[0]) の大きさが $size（$($res[1])）"; exit 1 }
        if (-not (Test-Path ([System.IO.Path]::ChangeExtension($png[0].FullName, '.json')))) { Write-Host "${id}: $($res[0]) の JSON が無い"; exit 1 }
    }
    Write-Host "${id}: 3 枚 OK（$shots）"
}
if (git status --porcelain -- Packages ProjectSettings) { Write-Host '書き出しで Packages・ProjectSettings が変わった（設計書 1.8・8.2）'; exit 1 }
exit 0

## 先に決めておくこと

- 期待値とテストの名前は設計書 9 章の表のまま。設計と違うことをしたくなったら、設計書を直さず（守るファイル）、報告の「やらなかったこと」に理由を書く。
- テストのファイル名はクラス名と同じ（設計書 9.3）。この段で書くテストの `.cs` は、次の段から守られる。
- 新しい `.cs` には頭書きを付ける（スキル `file-header`）。新しいファイルの `.meta` は Unity に作らせ、本体と一緒にコミットする。
- Unity エディタを窓で起動しない（バッチモードだけ）。
- `build-scape-exe.ps1` の `-Id` は各要素をさらに `,` で分ける（`pwsh -File` ではカンマの並びが 1 つの文字列で届く。8.1）。最後の行は `Result: PASS|FAIL|BLOCKED`（`unity-test.ps1` と同じ形）。
- 書き出しで `ProjectSettings/` が変わるなら、元に戻す方法を足す（1.8）。
