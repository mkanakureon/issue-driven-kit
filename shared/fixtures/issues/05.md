```pwsh
[Console]::OutputEncoding = [System.Text.Encoding]::UTF8
# 守る: Docs/Guides/Working/scape-m1-detailed-design.md
# 守る: Docs/Guides/Working/scape-tool-basic-design.md
# 守る: Assets/Tests/EditMode/SmokeTests.cs
$upTo = 'S1'
pwsh -NoProfile -File Scripts/unity-test.ps1 -Filter 'KaedeScapeBuii.Tests'
if ($LASTEXITCODE) { Write-Host 'EditMode のテストが通らない'; exit 1 }
$doc = Get-Content Docs/Guides/Working/scape-m1-detailed-design.md -Raw -Encoding utf8
$order = @('S1', 'S2', 'S4', 'S6a', 'S6b')
$rows = @([regex]::Matches($doc, '(?m)^\| `KaedeScapeBuii\.Tests\.(EditMode|PlayMode)\.(\w+)\.(\w+)` \| (S1|S2|S4|S6a|S6b) \|'))
if ($rows.Count -ne 54) { Write-Host "設計書 9 章のテストの行が $($rows.Count) 個（54 個のはず）"; exit 1 }
$need = @($rows | Where-Object { $order.IndexOf($_.Groups[4].Value) -le $order.IndexOf($upTo) })
$x = Get-ChildItem Logs/UnityTest -Directory | Sort-Object Name | Select-Object -Last 1
$xml = @{}
foreach ($p in @($need | ForEach-Object { $_.Groups[1].Value } | Sort-Object -Unique) + 'EditMode') {
    if ($xml.ContainsKey($p)) { continue }
    $f = Join-Path $x.FullName "$p-results.xml"
    if (-not (Test-Path $f)) { Write-Host "結果の XML が無い: $f"; exit 1 }
    $xml[$p] = [xml](Get-Content $f -Raw -Encoding utf8)
}
function Get-Cases([string]$plat, [string]$n) { @($xml[$plat].SelectNodes("//test-case[@fullname='$n' or starts-with(@fullname,'$n(')]")) }
foreach ($m in $need) {
    $plat = $m.Groups[1].Value
    $n = "KaedeScapeBuii.Tests.$plat.$($m.Groups[2].Value).$($m.Groups[3].Value)"
    $hit = @(Get-Cases $plat $n)
    if (-not $hit.Count) { Write-Host "テストが無い: $n"; exit 1 }
    if (@($hit | Where-Object { $_.result -ne 'Passed' }).Count) { Write-Host "通っていない: $n"; exit 1 }
}
$smoke = @(Get-Cases 'EditMode' 'KaedeScapeBuii.Tests.SmokeTests.TestRunnerWorks')
if (-not $smoke.Count -or $smoke[0].result -ne 'Passed') { Write-Host 'SmokeTests が通っていない'; exit 1 }
# [TestCase] の場合の数（9 章の表のとおり）
$cases = [ordered]@{
    'S1|EditMode|FocalLengthTableTests.HorizontalFovMatchesBasicDesignTable' = 5
    'S2|PlayMode|ShotCaptureTests.PngSizeMatchesResolution'                 = 3
    'S6a|PlayMode|TestWorldWalkTests.ForwardOneSecondMovesAtLeastOneMeter'  = 2
    'S6a|PlayMode|TestWorldShotTests.ThreeResolutionsInEachWorld'           = 2
}
foreach ($k in $cases.Keys) {
    $s, $plat, $m = $k -split '\|'
    if ($order.IndexOf($s) -gt $order.IndexOf($upTo)) { continue }
    $c = @(Get-Cases $plat "KaedeScapeBuii.Tests.$plat.$m").Count
    if ($c -lt $cases[$k]) { Write-Host "${m} の場合が $c 個（$($cases[$k]) 個のはず）"; exit 1 }
}
Write-Host "テストの名前 OK: $($need.Count) 本（S1〜$upTo）＋ SmokeTests"
if (git status --porcelain -- Packages ProjectSettings) { Write-Host 'Packages・ProjectSettings を変えている'; exit 1 }
exit 0
```
